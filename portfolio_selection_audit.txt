 261:

 263:         return None
 264: 
 265:     if isinstance(value, (pd.Timestamp, date)):
 266:         return value.isoformat()
 267:

 270: 
 271:     if isinstance(value, float) and not math.isfinite(value):
 272:         return None
 273: 
 274:     if isinstance(value, dict):
 275:         return {str(k): _clean(v) for k, v in value.items()}
 276: 
 277:     if isinstance(value, (list, tuple)):
 278:         return [_clean(v) for v in value]
 279: 
 280:     return value
