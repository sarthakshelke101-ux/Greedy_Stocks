"""Greedy V2 â€” sequential portfolio selection engine.

This module selects stocks using ONLY the price history supplied to it.
The caller is responsible for passing training-period prices, never
validation or test-period prices.

Method:
- Equal-weight portfolio.
- Sequential greedy selection.
- Maximise historical annualised Sharpe ratio.
- Maximum 10 stocks.
- Maximum 3 stocks per sector.
- Risk-free rate defaults to 0%.
"""

from __future__ import annotations

import math
from typing import Any

import sys
from pathlib import Path

import pandas as pd

ANALYTICS_DIR = Path(__file__).resolve().parent

if str(ANALYTICS_DIR) not in sys.path:
    sys.path.insert(0, str(ANALYTICS_DIR))

from portfolio_analytics import calculate_portfolio


def greedy_select_stocks(
    train_prices: pd.DataFrame,
    sectors: dict[str, str],
    target_stocks: int = 10,
    max_per_sector: int = 3,
    risk_free_rate: float = 0.0,
    fixed_symbols: list[str] | None = None,
) -> dict[str, Any]:
    """Select stocks sequentially using training-period Sharpe ratio."""

    if not isinstance(train_prices, pd.DataFrame):
        raise TypeError("train_prices must be a pandas DataFrame.")

    if train_prices.empty:
        raise ValueError("Training price data is empty.")

    if not isinstance(train_prices.index, pd.DatetimeIndex):
        raise ValueError("Training prices require a DatetimeIndex.")

    if train_prices.index.tz is not None:
        raise ValueError("Training dates must not contain a timezone.")

    if train_prices.index.has_duplicates:
        raise ValueError("Duplicate training dates detected.")

    if not train_prices.columns.is_unique:
        raise ValueError("Duplicate stock symbols detected.")

    if len(train_prices) < 60:
        raise ValueError("At least 60 price observations are required.")

    if not isinstance(target_stocks, int) or isinstance(target_stocks, bool):
        raise ValueError("target_stocks must be an integer.")

    if not isinstance(max_per_sector, int) or isinstance(max_per_sector, bool):
        raise ValueError("max_per_sector must be an integer.")

    if target_stocks < 1:
        raise ValueError("target_stocks must be at least 1.")

    if max_per_sector < 1:
        raise ValueError("max_per_sector must be at least 1.")

    if target_stocks > len(train_prices.columns):
        raise ValueError("Target exceeds the available number of stocks.")

    if not math.isfinite(float(risk_free_rate)):
        raise ValueError("risk_free_rate must be finite.")

    if risk_free_rate < 0:
        raise ValueError("risk_free_rate cannot be negative.")

    prices = train_prices.copy().sort_index()

    prices = prices.apply(pd.to_numeric, errors="coerce")

    if prices.isna().any().any():
        raise ValueError(
            "Training prices contain missing or invalid values. "
            "Do not fill or silently remove them."
        )

    if (prices <= 0).any().any():
        raise ValueError("Training prices must be positive.")

    symbols = list(prices.columns)

    missing_sectors = [
        symbol
        for symbol in symbols
        if symbol not in sectors
        or not isinstance(sectors[symbol], str)
        or not sectors[symbol].strip()
    ]

    if missing_sectors:
        raise ValueError(
            f"Missing sector classifications: {missing_sectors}"
        )

    sector_capacity = sum(
        min(
            max_per_sector,
            sum(
                sectors[symbol] == sector
                for symbol in symbols
            ),
        )
        for sector in set(sectors.values())
    )

    if target_stocks > sector_capacity:
        raise ValueError(
            "Target cannot be reached under the sector constraint."
        )

    # User-selected stocks are fixed starting holdings.
    fixed_symbols = [
        str(symbol).strip().upper()
        for symbol in (fixed_symbols or [])
    ]

    if len(fixed_symbols) != len(set(fixed_symbols)):
        raise ValueError("Duplicate user-selected stocks are not allowed.")

    unknown_symbols = [symbol for symbol in fixed_symbols if symbol not in symbols]
    if unknown_symbols:
        raise ValueError(
            f"User-selected stocks are unavailable in the eligible data: {unknown_symbols}"
        )

    if len(fixed_symbols) > target_stocks:
        raise ValueError(
            f"You selected {len(fixed_symbols)} stocks but requested "
            f"{target_stocks}. Increase portfolio size or remove selections."
        )

    fixed_sector_counts: dict[str, int] = {}
    for symbol in fixed_symbols:
        sector = sectors[symbol]
        fixed_sector_counts[sector] = fixed_sector_counts.get(sector, 0) + 1

    conflicts = {
        sector: count
        for sector, count in fixed_sector_counts.items()
        if count > max_per_sector
    }
    if conflicts:
        raise ValueError(
            f"Your selected stocks exceed the sector limit of "
            f"{max_per_sector}: {conflicts}"
        )

    selected: list[str] = list(fixed_symbols)
    selection_log: list[dict[str, Any]] = []

    print("\nGREEDY STOCKS â€” V2 SELECTION ENGINE")
    print("=" * 65)
    print("Available candidates:", len(symbols))
    print("Target portfolio size:", target_stocks)
    print("Maximum stocks per sector:", max_per_sector)
    print("Risk-free rate:", risk_free_rate)
    print("Training observations:", len(prices))
    print("Training dates:", prices.index.min().date(), "to", prices.index.max().date())

    # Sequential greedy selection.
    for step in range(len(selected) + 1, target_stocks + 1):

        candidates: list[dict[str, Any]] = []

        for symbol in symbols:

            if symbol in selected:
                continue

            current_sector_count = sum(
                sectors[item] == sectors[symbol]
                for item in selected
            )

            if current_sector_count >= max_per_sector:
                continue

            trial_symbols = selected + [symbol]

            result = calculate_portfolio(
                prices=prices,
                sectors=sectors,
                symbols=trial_symbols,
                risk_free_rate=risk_free_rate,
                max_stocks=target_stocks,
                max_per_sector=max_per_sector,
            )

            sharpe = result["metrics"].get("sharpe_ratio")

            if sharpe is None or pd.isna(sharpe):
                continue

            sharpe = float(sharpe)

            if not math.isfinite(sharpe):
                continue

            candidates.append({
                "symbol": symbol,
                "sector": sectors[symbol],
                "portfolio_sharpe": sharpe,
                "portfolio_size": len(trial_symbols),
            })

        if not candidates:
            raise ValueError(
                f"No valid candidates available at selection step {step}."
            )

        # Deterministic tie-breaking by ticker.
        candidates.sort(
            key=lambda item: (
                -item["portfolio_sharpe"],
                item["symbol"],
            )
        )

        winner = candidates[0]

        previous_sharpe = (
            selection_log[-1]["portfolio_sharpe"]
            if selection_log
            else None
        )

        selected.append(winner["symbol"])

        selection_log.append({
            "step": step,
            "symbol": winner["symbol"],
            "sector": winner["sector"],
            "portfolio_size": len(selected),
            "portfolio_sharpe": winner["portfolio_sharpe"],
            "previous_portfolio_sharpe": previous_sharpe,
            "marginal_sharpe_change": (
                winner["portfolio_sharpe"] - previous_sharpe
                if previous_sharpe is not None
                else None
            ),
            "candidates_evaluated": len(candidates),
        })

        print(
            f"Step {step:02d}: {winner['symbol']:<15} "
            f"Sector: {winner['sector']:<25} "
            f"Sharpe: {winner['portfolio_sharpe']:.4f}"
        )

    # Calculate final portfolio metrics.
    final_result = calculate_portfolio(
        prices=prices,
        sectors=sectors,
        symbols=selected,
        risk_free_rate=risk_free_rate,
        max_stocks=target_stocks,
        max_per_sector=max_per_sector,
    )

    print("\nSELECTION COMPLETE")
    print("-" * 65)
    print("Selected stocks:", len(selected))
    print("Unique sectors:", len({
        sectors[symbol] for symbol in selected
    }))
    print("Selected symbols:", ", ".join(selected))
    print("Final Sharpe:", final_result["metrics"].get("sharpe_ratio"))

    return {
        "selected_symbols": selected,
        "selection_log": pd.DataFrame(selection_log),
        "portfolio_result": final_result,
        "assumptions": {
            "selection_method": "SEQUENTIAL_GREEDY_SHARPE",
            "price_basis": "SPLIT_BONUS_ADJUSTED",
            "weighting": "EQUAL_WEIGHT",
            "daily_rebalancing": True,
            "risk_free_rate": risk_free_rate,
            "target_stocks": target_stocks,
            "user_selected_symbols": fixed_symbols,
            "greedy_selected_symbols": [
                x for x in selected if x not in fixed_symbols
            ],
            "max_per_sector": max_per_sector,
            "selection_uses_training_data_only": True,
            "globally_optimal": False,
        },
    }

from pathlib import Path
from datetime import date

import pandas as pd

from analytics.database_adapter import get_all_companies_with_prices
from analytics.adjusted_price_adapter import get_adjusted_company_prices
from analytics.index_adapter import get_index_closes


MINIMUM_OBSERVATIONS = 252
START_DATE = "2016-11-08"
END_DATE = date.today().isoformat()


class GreedyV2Dataset:
    def __init__(
        self,
        constituents,
        sectors,
        raw_prices,
        adjusted_prices,
        benchmark_prices,
        metadata,
    ):
        self.constituents = constituents
        self.sectors = sectors
        self.raw_prices = raw_prices
        self.adjusted_prices = adjusted_prices
        self.benchmark_prices = benchmark_prices
        self.metadata = metadata


def _normalise_price_series(series, name):
    result = pd.to_numeric(series, errors="coerce").copy()

    if not isinstance(result.index, pd.DatetimeIndex):
        result.index = pd.to_datetime(result.index, errors="raise")

    if result.index.tz is not None:
        raise ValueError(f"{name} contains timezone-aware dates.")

    result.index = result.index.normalize()
    result = result.sort_index()

    if result.index.has_duplicates:
        raise ValueError(f"{name} contains duplicate dates.")

    if result.isna().any():
        raise ValueError(f"{name} contains missing or invalid prices.")

    if (result <= 0).any():
        raise ValueError(f"{name} contains non-positive prices.")

    result.name = name
    return result


def _load_benchmark(start_date, end_date):
    benchmark = get_index_closes(start_date, end_date)

    if not isinstance(benchmark, pd.Series):
        raise TypeError("get_index_closes must return a pandas Series.")

    return _normalise_price_series(benchmark, "NIFTY_50")


def prepare_greedy_v2_dataset(
    start_date=START_DATE,
    end_date=END_DATE,
):
    print("\nGREEDY STOCKS â€” V2 DATA PREPARATION")
    print("=" * 65)

    constituents = get_all_companies_with_prices(
        min_observations=MINIMUM_OBSERVATIONS
    ).copy()

    required = {"company_id", "symbol", "sector"}
    missing = required - set(constituents.columns)

    if missing:
        raise ValueError(f"Missing constituent columns: {sorted(missing)}")

    if constituents["symbol"].duplicated().any():
        raise ValueError("Duplicate constituent symbols detected.")

    constituents = constituents.sort_values("symbol").reset_index(drop=True)

    sectors = dict(zip(constituents["symbol"], constituents["sector"]))

    raw_series = {}
    adjusted_series = {}
    coverage = []
    skipped_symbols = []

    for _, company in constituents.iterrows():
        symbol = str(company["symbol"])
        company_id = int(company["company_id"])

        result = get_adjusted_company_prices(
            company_id=company_id,
            start_date=start_date,
            end_date=end_date,
        )

        if result.empty:
            skipped_symbols.append(symbol)
            print(f"SKIP {symbol}: no prices in requested window")
            continue

        dates = pd.DatetimeIndex(pd.to_datetime(result["trade_date"]))

        raw = pd.Series(
            result["close"].to_numpy(),
            index=dates,
            name=symbol,
        )

        adjusted = pd.Series(
            result["adjusted_close"].to_numpy(),
            index=dates,
            name=symbol,
        )

        raw_series[symbol] = _normalise_price_series(
            raw, f"{symbol}_RAW"
        )

        adjusted_series[symbol] = _normalise_price_series(
            adjusted, symbol
        )

        coverage.append({
            "symbol": symbol,
            "observations": len(result),
            "first_date": dates.min(),
            "last_date": dates.max(),
        })

    if not adjusted_series:
        raise ValueError("No eligible companies have prices in the requested window.")

    # Keep only companies with observations in this requested window.
    constituents = constituents[
        constituents["symbol"].isin(adjusted_series)
    ].copy().reset_index(drop=True)
    sectors = dict(zip(constituents["symbol"], constituents["sector"]))

    # Preserve each stock's complete available history.
    raw_prices = pd.concat(raw_series.values(), axis=1).sort_index()
    adjusted_prices = pd.concat(
        adjusted_series.values(), axis=1
    ).sort_index()

    # Benchmark remains independent. No all-50-stock intersection.
    benchmark_prices = _load_benchmark(start_date, end_date)

    # Retain the full date range and missing values.
    # Downstream training/evaluation selects valid observations.
    metadata = {
        "requested_start_date": start_date,
        "requested_end_date": end_date,
        "actual_start_date": str(adjusted_prices.index.min().date()),
        "actual_end_date": str(adjusted_prices.index.max().date()),
        "constituent_count": len(adjusted_series),
        "skipped_no_window_history": skipped_symbols,
        "universe_basis": "ALL_COMPANIES_WITH_PRICE_HISTORY",
        "minimum_observations": MINIMUM_OBSERVATIONS,
        "inactive_companies_included": True,
        "price_observation_count": len(adjusted_prices),
        "benchmark_observation_count": len(benchmark_prices),
        "stock_symbols": list(adjusted_prices.columns),
        "price_basis": "SPLIT_BONUS_ADJUSTED",
        "raw_price_basis": "RAW_CLOSE_UNADJUSTED",
        "dividend_adjusted": False,
        "corporate_action_coverage": "PARTIAL",
        "membership_basis": "NOT_RESTRICTED_TO_INDEX_MEMBERSHIP",
        "session_reference": "Available stock-price observations",
        "forward_filled": False,
        "database_writes": False,
        "common_all_stock_dates": False,
        "company_coverage": coverage,
    }

    print("\nDATA QUALITY REPORT")
    print("-" * 65)
    print("Requested:", start_date, "to", end_date)
    print("Actual stock history:", metadata["actual_start_date"], "to", metadata["actual_end_date"])
    print("Eligible companies:", len(adjusted_prices.columns))
    print("Inactive companies included:", int((constituents["is_active"] == 0).sum()))
    print("Minimum observations:", MINIMUM_OBSERVATIONS)
    print("Skipped (no window history):", skipped_symbols)
    print("Adjusted matrix:", adjusted_prices.shape)
    print("Raw matrix:", raw_prices.shape)
    print("Benchmark observations:", len(benchmark_prices))
    print("Forward filling: False")
    print("Database writes: False")
    print("\nDATA PREPARATION: PASS")

    return GreedyV2Dataset(
        constituents=constituents,
        sectors=sectors,
        raw_prices=raw_prices,
        adjusted_prices=adjusted_prices,
        benchmark_prices=benchmark_prices,
        metadata=metadata,
    )


if __name__ == "__main__":
    dataset = prepare_greedy_v2_dataset()

    print("\nFINAL MATRIX SHAPES")
    print("Raw:", dataset.raw_prices.shape)
    print("Adjusted:", dataset.adjusted_prices.shape)

"""Chronological train/validation/test splits for Greedy V2."""

from dataclasses import dataclass
import numpy as np
import pandas as pd


@dataclass(frozen=True)
class DataSplits:
    train: pd.DataFrame
    validation: pd.DataFrame
    test: pd.DataFrame
    metadata: dict


def chronological_split(
    prices: pd.DataFrame,
    train_fraction: float = 0.60,
    validation_fraction: float = 0.20,
    allow_sparse_evaluation: bool = False,
) -> DataSplits:
    """Chronological split; optionally exclude incomplete training candidates.

    In sparse evaluation mode, a symbol must have complete training prices.
    Missing values in validation/test are preserved and never filled.
    """

    if not isinstance(prices, pd.DataFrame):
        raise TypeError("prices must be a pandas DataFrame")
    if prices.empty or len(prices) < 60:
        raise ValueError("At least 60 price observations are required")
    if not isinstance(prices.index, pd.DatetimeIndex):
        raise ValueError("prices must have a DatetimeIndex")
    if prices.index.has_duplicates:
        raise ValueError("Duplicate dates are not allowed")
    if not prices.index.is_monotonic_increasing:
        raise ValueError("Price dates must be sorted ascending")
    if not prices.columns.is_unique:
        raise ValueError("Duplicate price columns are not allowed")
    if not 0 < train_fraction < 1:
        raise ValueError("train_fraction must be between 0 and 1")
    if not 0 < validation_fraction < 1:
        raise ValueError("validation_fraction must be between 0 and 1")
    if train_fraction + validation_fraction >= 1:
        raise ValueError("Train and validation fractions must total less than 1")

    try:
        numeric = prices.apply(pd.to_numeric, errors="raise")
    except (TypeError, ValueError) as exc:
        raise ValueError("Prices must contain numeric values or missing values") from exc

    values = numeric.to_numpy(dtype=float)
    observed_mask = ~np.isnan(values)
    observed_values = values[observed_mask]

    if not np.isfinite(observed_values).all():
        raise ValueError("Observed prices must be finite")
    if (observed_values <= 0).any():
        raise ValueError("Observed prices must be positive")

    total = len(numeric)
    train_end = int(total * train_fraction)
    validation_end = int(total * (train_fraction + validation_fraction))

    if train_end < 2 or validation_end - train_end < 2:
        raise ValueError("Each split needs enough observations")
    if total - validation_end < 2:
        raise ValueError("Test split needs at least two observations")

    original_columns = list(numeric.columns)
    training_window = numeric.iloc[:train_end]
    eligible_columns = [
        col for col in original_columns
        if training_window[col].notna().all()
    ]
    excluded_columns = [
        col for col in original_columns if col not in eligible_columns
    ]

    if excluded_columns and not allow_sparse_evaluation:
        raise ValueError("Missing prices are not allowed")
    if not eligible_columns:
        raise ValueError("No stocks have complete prices throughout training")

    selected = numeric.loc[:, eligible_columns]
    train = selected.iloc[:train_end].copy()
    validation = selected.iloc[train_end - 1:validation_end].copy()
    test = selected.iloc[validation_end - 1:].copy()

    if train.isna().any().any():
        raise ValueError("Training prices must be complete")
    if train.empty or validation.empty or test.empty:
        raise ValueError("Insufficient historical data for all periods")

    metadata = {
        "method": "CHRONOLOGICAL",
        "total_observations": total,
        "train_observations": len(train),
        "validation_observations": len(validation) - 1,
        "test_observations": len(test) - 1,
        "train_start": str(train.index[0].date()),
        "train_end": str(train.index[-1].date()),
        "validation_start": str(validation.index[1].date()),
        "validation_end": str(validation.index[-1].date()),
        "test_start": str(test.index[1].date()),
        "test_end": str(test.index[-1].date()),
        "candidate_columns_before_filter": len(original_columns),
        "eligible_training_symbols": eligible_columns,
        "excluded_incomplete_training_symbols": excluded_columns,
        "baseline_prices_included": True,
        "shuffled": False,
        "missing_prices_filled": False,
        "sparse_evaluation_allowed": allow_sparse_evaluation,
    }

    return DataSplits(train, validation, test, metadata)

"""Compare Greedy V2 against equal-weight and random baselines."""

ROOT = Path(__file__).resolve().parents[1]
OUTPUT_DIR = ROOT / "database" / "processed" / "analytics"

TARGET_STOCKS = 10
MAX_PER_SECTOR = 3
RISK_FREE_RATE = 0.0
PRICE_BASIS = "SPLIT_BONUS_ADJUSTED"
RETURN_TYPE = "PRICE_RETURN"

RANDOM_PORTFOLIO_COUNT = 500
RANDOM_SEED = 42


def safe_value(value):
    if isinstance(value, dict):
        return {str(k): safe_value(v) for k, v in value.items()}
    if isinstance(value, (list, tuple)):
        return [safe_value(v) for v in value]
    if isinstance(value, (np.integer, np.floating)):
        value = value.item()
    if isinstance(value, float) and not np.isfinite(value):
        return None
    if isinstance(value, pd.Timestamp):
        return value.isoformat()
    return value


def scalar_metrics(portfolio):
    metrics = portfolio["metrics"]
    keys = [
        "period_return_pct",
        "cagr_pct",
        "annualized_volatility_pct",
        "maximum_drawdown_pct",
        "sharpe_ratio",
    ]
    return {key: metrics.get(key) for key in keys}


def evaluate_portfolio(
    prices,
    benchmark,
    sectors,
    symbols,
    max_stocks,
    max_per_sector,
):
    portfolio = calculate_portfolio(
        prices=prices,
        sectors=sectors,
        symbols=symbols,
        risk_free_rate=RISK_FREE_RATE,
        max_stocks=max_stocks,
        max_per_sector=max_per_sector,
        price_basis=PRICE_BASIS,
        return_type=RETURN_TYPE,
    )

    comparison = compare_with_benchmark(
        portfolio["daily_returns"],
        benchmark,
    )

    return {
        "symbols": list(symbols),
        "metrics": scalar_metrics(portfolio),
        "portfolio_return_pct": comparison["portfolio_return_pct"],
        "benchmark_return_pct": comparison["benchmark_return_pct"],
        "excess_return_pct": comparison["excess_return_pct"],
        "common_return_observations": comparison[
            "common_return_observations"
        ],
        "metadata": portfolio["metadata"],
    }


def generate_random_portfolios(
    symbols,
    sectors,
    count=RANDOM_PORTFOLIO_COUNT,
    seed=RANDOM_SEED,
):
    """Generate unique random portfolios respecting sector limits."""

    rng = random.Random(seed)
    symbols = list(symbols)

    portfolios = set()
    attempts = 0
    max_attempts = count * 10000

    while len(portfolios) < count and attempts < max_attempts:
        attempts += 1
        candidate = tuple(sorted(rng.sample(symbols, TARGET_STOCKS)))

        sector_counts = {}
        for symbol in candidate:
            sector = sectors[symbol]
            sector_counts[sector] = sector_counts.get(sector, 0) + 1

        if all(
            amount <= MAX_PER_SECTOR
            for amount in sector_counts.values()
        ):
            portfolios.add(candidate)

    if len(portfolios) != count:
        raise RuntimeError(
            f"Generated only {len(portfolios)} of {count} "
            "requested unique valid portfolios."
        )

    return sorted(portfolios)


def summarize_random_results(rows):
    """Summarize the distribution of random portfolio outcomes."""

    frame = pd.DataFrame(rows)

    summary = {}

    columns = [
        "portfolio_return_pct",
        "excess_return_pct",
        "sharpe_ratio",
        "annualized_volatility_pct",
        "maximum_drawdown_pct",
    ]

    for column in columns:
        values = pd.to_numeric(
            frame[column],
            errors="coerce",
        ).dropna()

        summary[column] = {
            "mean": float(values.mean()),
            "median": float(values.median()),
            "p05": float(values.quantile(0.05)),
            "p95": float(values.quantile(0.95)),
            "minimum": float(values.min()),
            "maximum": float(values.max()),
        }

    summary["portfolios_count"] = len(frame)

    return summary


def run_period_comparison(
    period_name,
    prices,
    benchmark,
    sectors,
    greedy_symbols,
    random_portfolios,
):
    print(f"\nBASELINE COMPARISON â€” {period_name}")
    print("-" * 65)

    # Greedy V2: fixed training-selected stocks.
    greedy = evaluate_portfolio(
        prices,
        benchmark,
        sectors,
        greedy_symbols,
        TARGET_STOCKS,
        MAX_PER_SECTOR,
    )

    # All 50 NIFTY constituents, equal weighted.

def greedy_select(train_prices, sectors):
    selected = []

    for step in range(TARGET_STOCKS):
        candidates = []

        for symbol in train_prices.columns:
            if symbol in selected:
                continue

            sector_count = sum(
                sectors[s] == sectors[symbol] for s in selected
            )

            if sector_count >= MAX_PER_SECTOR:
                continue

            trial = selected + [symbol]

            result = calculate_portfolio(
                train_prices,
                sectors,
                symbols=trial,
                risk_free_rate=0.0,
            )

            sharpe = result["metrics"].get("sharpe_ratio")

            if sharpe is not None and pd.notna(sharpe):
                candidates.append((float(sharpe), symbol))

        if not candidates:
            raise ValueError(f"No eligible candidate at step {step + 1}")

        candidates.sort(reverse=True)
        selected.append(candidates[0][1])

    return selected

COST_BPS_PER_TURNOVER = 10.0
MIN_TRAIN_OBS = 60
STRATEGIES = [
    "GREEDY_V2_BASELINE",
    "RISK_AWARE",
    "DIVERSIFICATION_AWARE",
    "COST_AWARE",
    "STABILITY_BASED",

def stats(returns: pd.Series) -> dict:
    r = pd.to_numeric(returns, errors="coerce").dropna().astype(float)
    if len(r) < 2:
        raise ValueError("At least two daily returns required.")
    wealth = (1 + r).cumprod()
    peak = wealth.cummax()
    dd = wealth / peak - 1
    vol = float(r.std(ddof=1) * np.sqrt(252))
    sharpe = float((r.mean() * 252) / vol) if vol > 0 else 0.0
    return {
        "total_return_pct": float((wealth.iloc[-1] - 1) * 100),
        "annualized_volatility_pct": vol * 100,
        "sharpe_ratio": sharpe,
        "maximum_drawdown_pct": float(dd.min() * 100),
        "observations": int(len(r)),
    }


def trial_returns(prices, sectors, symbols):
    p = calculate_portfolio(
        prices=prices, sectors=sectors, symbols=symbols,
        risk_free_rate=RISK_FREE_RATE, max_stocks=TARGET_STOCKS,
        max_per_sector=MAX_PER_SECTOR, price_basis="SPLIT_BONUS_ADJUSTED",
        return_type="PRICE_RETURN",
    )
    return p["daily_returns"]


def turnover_cost_adjusted_sharpe(returns: pd.Series, bps: float) -> tuple[float, float]:
    # Approximate daily-rebalance turnover from portfolio weights drifting
    # after returns. Cost is charged on one-way traded notional.
    r = returns.astype(float)
    wealth = 1.0
    weights = np.repeat(1.0 / TARGET_STOCKS, TARGET_STOCKS)
    # Equal-weight daily rebalance implies roughly half the sum of absolute
    # deviations from target after each session. Use portfolio-level proxy
    # based on cross-sectional dispersion unavailable in portfolio returns;
    # therefore use a transparent conservative daily turnover assumption.
    # At minimum, a full rebalance cost proxy is 1/N of portfolio notional.
    daily_turnover = 1.0 - 1.0 / TARGET_STOCKS
    cost = daily_turnover * (bps / 10000.0)
    net = r - cost
    return stats(net)["sharpe_ratio"], cost * 100


def strategy_score(prices, sectors, selected, candidate, strategy, cost_bps):
    trial = selected + [candidate]
    ret = trial_returns(prices, sectors, trial)
    m = stats(ret)
    if strategy == "RISK_AWARE":
        # Penalise both annualised volatility and peak-to-trough drawdown.
        score = m["sharpe_ratio"] - 0.015 * m["annualized_volatility_pct"] - 0.02 * abs(m["maximum_drawdown_pct"])
    elif strategy == "DIVERSIFICATION_AWARE":
        stock_ret = prices[trial].pct_change(fill_method=None).dropna(how="any")
        corr = stock_ret.corr().abs().to_numpy()
        upper = corr[np.triu_indices_from(corr, k=1)]
        mean_corr = float(upper.mean()) if len(upper) else 0.0
        score = m["sharpe_ratio"] - 0.75 * mean_corr
    elif strategy == "COST_AWARE":
        net_sharpe, _ = turnover_cost_adjusted_sharpe(ret, cost_bps)
        score = net_sharpe
    elif strategy == "STABILITY_BASED":
        # Score the same trial portfolio over three chronological slices
        # of training data; reward mean Sharpe and penalise dispersion.
        chunks = [prices[trial].iloc[idx].copy() for idx in np.array_split(np.arange(len(prices)), 3) if len(idx) >= MIN_TRAIN_OBS]
        if len(chunks) < 2:
            score = m["sharpe_ratio"]
        else:
            sh = [stats(trial_returns(chunk, sectors, trial))["sharpe_ratio"] for chunk in chunks]
            score = float(np.mean(sh) - 0.5 * np.std(sh, ddof=0))
    else:
        raise ValueError(strategy)
    return float(score), m


def experimental_select(prices, sectors, strategy, cost_bps=COST_BPS_PER_TURNOVER):
    symbols = list(prices.columns)
    selected = []
    logs = []
    for step in range(1, TARGET_STOCKS + 1):
        candidates = []
        for sym in symbols:
            if sym in selected:
                continue
            if sum(sectors[x] == sectors[sym] for x in selected) >= MAX_PER_SECTOR:
                continue
            score, metrics = strategy_score(prices, sectors, selected, sym, strategy, cost_bps)
            candidates.append((score, sym, metrics))
        if not candidates:
            raise ValueError(f"{strategy}: no eligible candidate at step {step}")
        candidates.sort(key=lambda x: (-x[0], x[1]))
        score, winner, metrics = candidates[0]
        selected.append(winner)
        logs.append({
            "strategy": strategy, "step": step, "symbol": winner,
            "sector": sectors[winner], "selection_score": score,
            "training_sharpe_at_selection": metrics["sharpe_ratio"],
        })
        print(f"  {strategy} step {step:02d}/{TARGET_STOCKS}: {winner} score={score:.4f}")
    return selected, logs


def evaluate(prices, benchmark, sectors, selected, label):

        "excess_return_pct": comparison["excess_return_pct"],
        "common_return_observations": comparison["common_return_observations"],
    }

def json_safe(value):
    """Convert supported objects into JSON-compatible values."""

    if isinstance(value, dict):
        return {
            str(key): json_safe(item)
            for key, item in value.items()
        }

    if isinstance(value, (list, tuple)):
        return [json_safe(item) for item in value]

    if isinstance(value, pd.Timestamp):
        return value.isoformat()

    if hasattr(value, "item"):
        return value.item()

    return value


def scalar_metrics(portfolio):
    """Return scalar financial metrics without pandas objects."""

    metrics = portfolio["metrics"]
    output = {}

    for key, value in metrics.items():
        if isinstance(value, (str, int, float, bool)) or value is None:
            output[key] = value

    return output

    print("Loading historical prices for 50 companies...")

    price_series = {}

    for _, row in constituents.iterrows():
        symbol = row["symbol"]
        series = get_company_prices(
            int(row["company_id"]),
            START_DATE,
            END_DATE,
        )
        price_series[symbol] = series.rename(symbol)

    all_prices = pd.concat(price_series.values(), axis=1).sort_index()

    # Fair comparison: every candidate uses exactly the same dates.
    prices = all_prices.dropna(axis=0, how="any")

    if len(prices) < 60:
        raise ValueError("Insufficient common price history.")

    if prices.isna().any().any():
        raise ValueError("Common price matrix still contains missing values.")

    print(f"\nCommon price observations: {len(prices)}")
    print(f"First date: {prices.index.min().date()}")
    print(f"Last date: {prices.index.max().date()}")
    print(f"Expected return observations: {len(prices) - 1}")

    selected = []
    selection_log = []

    print("\nSTARTING GREEDY SELECTION")
    print("-" * 65)

    for step in range(1, TARGET_STOCKS + 1):
        candidates = []

        for symbol in prices.columns:
            if symbol in selected:
                continue

            current_sector_count = sum(
                sectors[s] == sectors[symbol] for s in selected
            )

            if current_sector_count >= MAX_PER_SECTOR:
                continue

            trial = selected + [symbol]

            result = calculate_portfolio(
                prices=prices,
                sectors=sectors,
                symbols=trial,
                risk_free_rate=0.0,
            )

            sharpe = result["metrics"].get("sharpe_ratio")

            if sharpe is None or pd.isna(sharpe):
                continue

            candidates.append({
                "symbol": symbol,
                "sector": sectors[symbol],
                "sharpe": float(sharpe),
                "portfolio_size": len(trial),
            })

        if not candidates:
            raise ValueError(
                f"No valid candidates available at selection step {step}."
            )

        candidates.sort(
            key=lambda item: item["sharpe"],
            reverse=True,
        )

        winner = candidates[0]

        )

            f"Selected: {winner['symbol']:15} | "

"""Read-only source-code knowledge layer for Ask GREEDY."""
from pathlib import Path
import logging

logger = logging.getLogger("greedy_stocks_chat.knowledge")

ROOT = Path(__file__).resolve().parents[2]
ANALYTICS = ROOT / "analytics"

# Only trusted project source directories and explicit analytical modules.
MAX_FILES = 14
MAX_FILE_CHARS = 6500
MAX_TOTAL_CHARS = 28000

EXACT_FILES = [
    "greedy_v2_engine.py",
    "greedy_v2_data.py",
    "greedy_v2_splits.py",
    "financial_metrics.py",
    "portfolio_analytics.py",
    "portfolio_allocation.py",
]

def discover_sources():
    found = {}

    for name in EXACT_FILES:
        path = ANALYTICS / name
        if path.is_file():
            found[path.resolve()] = path

    # Discover versioned implementations without assuming their names.
    for path in ANALYTICS.rglob("*.py"):
        if not path.is_file():
            continue
        low = path.name.lower()
        if "greedy" in low and any(x in low for x in ("v1", "v2", "v3", "engine", "algorithm")):
            found[path.resolve()] = path

    # Also discover Greedy-named backend algorithm modules.
    backend = ROOT / "backend"
    if backend.exists():
        for path in backend.glob("*greedy*.py"):
            if path.is_file():
                found[path.resolve()] = path

    return sorted(found.values(), key=lambda p: str(p).lower())[:MAX_FILES]

def build_greedy_knowledge_context(question=""):
    """Return bounded source excerpts; never execute source code."""
    pieces = []
    total = 0

    for path in discover_sources():
        try:
            text = path.read_text(encoding="utf-8-sig")
        except (OSError, UnicodeError) as exc:
            logger.warning("Could not read algorithm source %s: %s", path.name, exc)
            continue

        # Keep context bounded and avoid exposing unrelated oversized modules.
        excerpt = text[:MAX_FILE_CHARS]
        relative = path.relative_to(ROOT).as_posix()
        block = (
            f"\n--- SOURCE FILE: {relative} ---\n"
            f"{excerpt}\n"
            f"--- END SOURCE FILE: {relative} ---\n"
        )

        if total + len(block) > MAX_TOTAL_CHARS:
            break

        pieces.append(block)
        total += len(block)

    if not pieces:
        return (
            "GREEDY SOURCE STATUS: No matching algorithm source files were found. "
            "Do not invent version-specific implementation details."
        )

    return (
        "GREEDY STOCKS SOURCE-CODE KNOWLEDGE\n"
        "The following excerpts are read-only project source files. "
        "Use them as evidence when explaining algorithm implementation.\n"
        "Do not claim a version exists unless its source is present. "
        "Do not infer undocumented behaviour. If a file is truncated, "
        "say the available excerpt does not establish the missing detail.\n"
        + "".join(pieces)
    )

def _extract_json(text):
    text = (text or "").strip()
    text = re.sub(r"^```(?:json)?\s*", "", text, flags=re.I)
    text = re.sub(r"\s*```$", "", text)

    start = text.find("{")
    end = text.rfind("}")

    if start < 0 or end <= start:
        raise ValueError("Planner returned no JSON object.")

    result = json.loads(text[start:end + 1])

    if not isinstance(result, dict):
        raise ValueError("Planner response must be an object.")

    return result


def validate_research_plan(plan):
    """Strictly validate a sequential dependency-aware operation plan."""

    if not isinstance(plan, dict):
        raise ValueError("Research plan must be an object.")

    if set(plan) != {"operations"}:
        raise ValueError("Unexpected research plan fields.")

    operations = plan["operations"]

    if not isinstance(operations, list):
        raise ValueError("Operations must be a list.")

    if not 1 <= len(operations) <= MAX_OPERATIONS:
        raise ValueError("Invalid number of research operations.")

    seen = set()
    validated = []
    market_count = 0

    for item in operations:
        if not isinstance(item, dict):
            raise ValueError("Operation must be an object.")

        if set(item) != {"id", "operation", "query", "depends_on"}:
            raise ValueError("Invalid operation schema.")

        operation_id = item["id"]
        operation = item["operation"]
        query = item["query"]
        dependencies = item["depends_on"]

        if not isinstance(operation_id, str):
            raise ValueError("Operation ID must be text.")

        if not re.fullmatch(r"[a-z][a-z0-9_]{0,39}", operation_id):
            raise ValueError("Invalid operation ID.")

        if operation_id in seen:
            raise ValueError("Duplicate operation ID.")

        if operation not in ALLOWED_OPERATIONS:
            raise ValueError("Unsupported operation.")

        if not isinstance(query, str):
            raise ValueError("Operation query must be text.")

        query = query.strip()

        if not query or len(query) > MAX_QUERY_LENGTH:
            raise ValueError("Invalid operation query.")

        if not isinstance(dependencies, list):
            raise ValueError("Dependencies must be a list.")

        if len(dependencies) != len(set(dependencies)):
            raise ValueError("Duplicate dependencies.")

        # Only earlier operations may be dependencies.
        if any(dep not in seen for dep in dependencies):
            raise ValueError("Dependency is missing or out of order.")

        if operation == "market_research":
            market_count += 1

        validated.append({
            "id": operation_id,
            "operation": operation,
            "query": query,
            "depends_on": list(dependencies),
        })

        seen.add(operation_id)

    if market_count != 1:
        raise ValueError("Exactly one market_research operation is required.")

    return validated

def build_research_plan(client, model, question, history):
    """Ask Gemini for a plan, validate it, and fail safely."""

    history_text = []

    for item in (history or [])[-10:]:
        if isinstance(item, dict):
            history_text.append({
                "role": str(item.get("role", ""))[:20],
                "content": str(item.get("content", ""))[:2000],
            })

    prompt = (
        PLANNER_INSTRUCTION
        + "\n\nCONVERSATION HISTORY:\n"
        + json.dumps(history_text, ensure_ascii=False)
        + "\n\nCURRENT QUESTION:\n"
        + str(question)[:5000]
    )

    try:
        interaction = client.interactions.create(
            model=model,
            input=prompt,
            system_instruction=(
                "You are a strict research planner. "
                "Return valid JSON only. Do not answer the user."
            ),
        )

        parsed = _extract_json(
            getattr(interaction, "output_text", "")
        )

        return validate_research_plan(parsed), "GEMINI_PLANNED"

    except Exception as exc:
        logger.warning(
            "Research planner fallback activated: %s",
            type(exc).__name__,
        )

        fallback = fallback_research_plan(question)

        return (
            validate_research_plan({"operations": fallback}),
            "DETERMINISTIC_FALLBACK",
        )


def execute_research_plan(plan, handlers):
    """Execute operations sequentially and respect dependencies."""

    results = {}
    execution_log = []

    for operation in plan:
        operation_id = operation["id"]
        name = operation["operation"]

        failed_dependencies = [
            dep for dep in operation["depends_on"]
            if results.get(dep, {}).get("status") != "COMPLETED"
        ]

        if failed_dependencies:
            result = {
                "status": "SKIPPED",
                "reason": "A required dependency did not complete.",
                "failed_dependencies": failed_dependencies,
            }

        else:
            handler = handlers.get(name)

            if handler is None:
                result = {
                    "status": "FAILED",
                    "reason": "No execution handler is registered.",
                }

            else:
                try:
                    value = handler(operation["query"])

                    if value is None:
                        raise ValueError("Operation returned no result.")

                    result = {
                        "status": "COMPLETED",
                        "result": value,
                    }

                except Exception as exc:
                    logger.exception(
                        "Research operation %s failed.",
                        operation_id,
                    )

                    result = {
                        "status": "FAILED",
                        "reason": (
                            f"{type(exc).__name__}: {str(exc)[:300]}"
                        ),
                    }

        results[operation_id] = result

        execution_log.append({
            "id": operation_id,
            "operation": name,
            **result,
        })

    return execution_log


def format_execution_context(plan, planning_status, execution_log):
    """Prepare bounded, transparent research evidence for Gemini."""

    payload = {
        "planning_status": planning_status,
        "operation_plan": [
            {
                "id": item["id"],
                "operation": item["operation"],
                "depends_on": item["depends_on"],
            }
            for item in plan
        ],
        "execution": execution_log,
    }

    rendered = json.dumps(
        payload,
        ensure_ascii=False,
        default=str,
        indent=2,
    )

    return (
        "PHASE 2 RESEARCH PLAN AND EXECUTION TRACE\n"
        "The following records show which operations were actually run.\n"
        "COMPLETED means the handler returned a result; it does not "
        "automatically certify every underlying data value.\n"
        "FAILED and SKIPPED operations must be disclosed when relevant.\n\n"
        + rendered[:40000]
    )

def _json(text_value):
    s = (text_value or "").strip()
    s = re.sub(r"^```(?:json)?|```$", "", s, flags=re.I).strip()
    a, b = s.find("{"), s.rfind("}")
    if a < 0 or b <= a:
        raise ValueError("Invalid planner JSON")
    obj = json.loads(s[a:b + 1])
    if not isinstance(obj, dict):
        raise ValueError("Planner output is not an object")
    return obj


def _plan(client, model, question):
    today = date.today()
    response = client.models.generate_content(
        model=model,
        contents=f"{PLANNER}\nToday: {today}\nQuestion: {question}",
    )
    p = _json(getattr(response, "text", ""))
    symbol = p.get("symbol")
    if symbol:
        symbol = str(symbol).strip().upper()
        if not re.fullmatch(r"[A-Z0-9&._-]{1,30}", symbol):
            raise ValueError("Invalid ticker format")

    end = date.fromisoformat(p["end_date"]) if p.get("end_date") else today
    start = (
        date.fromisoformat(p["start_date"])
        if p.get("start_date")
        else end - timedelta(days=365 * MAX_YEARS + 5)
    )
    earliest = today - timedelta(days=365 * MAX_YEARS + 5)
    start, end = max(start, earliest), min(end, today)
    if start > end:
        raise ValueError("Invalid date range")

    return (
        symbol,
        start,
        end,
        bool(p.get("include_nifty")),
        str(p.get("analysis_request") or question)[:1000],
    )


def _stock(symbol, start, end):
    # Restrict analysis to the latest available NIFTY 50 snapshot.
    constituents = database_adapter.get_latest_constituents()
    supported = set(
        constituents["symbol"].astype(str).str.strip().str.upper()
    )
    if symbol not in supported:
        raise ValueError(
            f"{symbol} is not in the latest available NIFTY 50 constituent snapshot."
        )

    query = text("""
        SELECT id, symbol, company_name
        FROM companies
        WHERE UPPER(symbol) = :symbol
        LIMIT 1
    """)
    with engine.connect() as conn:
        company = conn.execute(
            query, {"symbol": symbol}
        ).mappings().first()

    if not company:
        raise ValueError(f"Company master record not found for {symbol}.")

    frame = adjusted_price_adapter.get_adjusted_company_prices(
        company_id=int(company["id"]),
        start_date=start.isoformat(),
        end_date=end.isoformat(),
    )
    if frame.empty or "adjusted_close" not in frame.columns:
        raise ValueError(f"No adjusted price observations found for {symbol}.")

    frame = frame.copy()
    frame["trade_date"] = pd.to_datetime(frame["trade_date"])
    frame["close_price"] = pd.to_numeric(
        frame["adjusted_close"], errors="coerce"
    )
    frame = frame.dropna(subset=["trade_date", "close_price"])
    frame = frame[frame["close_price"] > 0]
    frame = frame.sort_values("trade_date").drop_duplicates("trade_date")

    if len(frame) < 2:
        raise ValueError(f"Insufficient adjusted price history for {symbol}.")

    frame["symbol"] = str(company["symbol"])
    frame["company_name"] = str(company["company_name"])
    return frame.reset_index(drop=True)


def _nifty(start, end):
    series = index_adapter.get_index_closes(
        start_date=start.isoformat(),
        end_date=end.isoformat(),
    )
    if series is None or series.empty:
        return pd.DataFrame(columns=["trade_date", "close_value"])

    frame = series.rename("close_value").rename_axis("trade_date").reset_index()
    frame["trade_date"] = pd.to_datetime(frame["trade_date"])
    frame["close_value"] = pd.to_numeric(
        frame["close_value"], errors="coerce"
    )
    return (
        frame.dropna()
        .query("close_value > 0")
        .sort_values("trade_date")
        .drop_duplicates("trade_date")
        .reset_index(drop=True)
    )


def _year_returns(prices):
    result = []
    for year, group in prices.groupby(prices.index.year):
        if len(group) < 2:
            continue
        result.append({
            "year": int(year),
            "first_observation": group.index[0].date().isoformat(),
            "last_observation": group.index[-1].date().isoformat(),
            "start_adjusted_close": round(float(group.iloc[0]), 4),
            "end_adjusted_close": round(float(group.iloc[-1]), 4),
            "return_pct": round(
                float((group.iloc[-1] / group.iloc[0] - 1) * 100), 4
            ),
            "observations": int(len(group)),
        })
    return result


def _analyse(df, nifty, request):
    prices = (
        df.set_index("trade_date")["close_price"]
        .astype(float)
        .sort_index()
    )
    prices = prices[~prices.index.duplicated(keep="last")]

    if len(prices) < 2:
        raise ValueError("Insufficient observations")

    metrics = calculate_metrics(
        prices,
        risk_free_rate=0.0,
        price_basis="SPLIT_BONUS_ADJUSTED",
        return_type="PRICE_RETURN",
    )

    peak = prices.cummax()
    drawdown = prices / peak - 1
    trough = drawdown.idxmin()
    peak_date = prices.loc[:trough].idxmax()
    calendar = _year_returns(prices)

    result = {
        "company": str(df.iloc[0]["company_name"]),
        "symbol": str(df.iloc[0]["symbol"]),
        "request": request,
        "price_basis": (
            "Split/bonus-adjusted close prices; dividends are not included."
        ),
        "coverage": {
            "first_date": metrics["start_date"],
            "last_date": metrics["end_date"],
            "price_observations": metrics["price_observations"],
            "daily_return_observations": metrics["return_observations"],
        },
        "period_return_pct": metrics["period_return_pct"],
        "cagr_pct": metrics["cagr_pct"],
        "annualized_volatility_pct": metrics["annualized_volatility_pct"],
        "sharpe_ratio": metrics["sharpe_ratio"],
        "sharpe_assumptions": {
            "annual_risk_free_rate": 0.0,
            "trading_days_per_year": 252,
            "method": "Annualized daily excess-return Sharpe ratio",
        },
        "maximum_drawdown": {
            "pct": round(float(drawdown.min() * 100), 4),
            "peak_date": peak_date.date().isoformat(),
            "trough_date": trough.date().isoformat(),
        },
        "calendar_year_returns": calendar,
        "best_calendar_year": (
            max(calendar, key=lambda x: x["return_pct"]) if calendar else None
        ),
        "worst_calendar_year": (
            min(calendar, key=lambda x: x["return_pct"]) if calendar else None
        ),
        "metric_methodology": {
            "trading_days_per_year": metrics["trading_days_per_year"],
            "risk_free_rate_annual": metrics["risk_free_rate_annual"],
            "calendar_validation": metrics["calendar_validation"],
            "cagr_status": metrics["cagr_status"],
        },
    }

    if nifty is not None and not nifty.empty:
        benchmark = nifty.set_index("trade_date")["close_value"].astype(float)
        aligned = pd.concat(
            [prices.rename("stock"), benchmark.rename("nifty")],
            axis=1,
            join="inner",
        ).dropna()

        if len(aligned) >= 2:
            result["nifty_comparison"] = {
                "matched_observations": int(len(aligned)),
                "first_matched_date": aligned.index[0].date().isoformat(),
                "last_matched_date": aligned.index[-1].date().isoformat(),
                "stock_return_pct": round(
                    float((aligned.stock.iloc[-1] / aligned.stock.iloc[0] - 1) * 100),
                    4,
                ),
                "nifty_return_pct": round(
                    float((aligned.nifty.iloc[-1] / aligned.nifty.iloc[0] - 1) * 100),
                    4,
                ),
                "benchmark_basis": "NIFTY 50 index close price return",
            }
        else:
            result["nifty_comparison"] = {
                "status": "UNAVAILABLE",
                "reason": "Insufficient matched stock and index observations.",
            }

    return result

        return (

    except Exception as exc:

        return (

            f"Reason: {type(exc).__name__}: {str(exc)[:400]}\n"

    user_message: str
    assistant_message: str
    created_at: datetime

def get_or_create_dev_user(connection):
    user_id = connection.execute(
        text("""
            SELECT id
            FROM app_users
            WHERE email = :email
        """),
        {"email": DEV_USER_EMAIL},
    ).scalar_one_or_none()

    if user_id is not None:
        return user_id

    result = connection.execute(
        text("""
            INSERT INTO app_users (
                username,
                email,
                preferred_language,
                experience_level
            )
            VALUES (
                'GREEDY Development User',
                :email,
                'English',
                'Complete Beginner'
            )
        """),
        {"email": DEV_USER_EMAIL},
    )

    return result.lastrowid

def get_recent_history(conversation_id):
    if conversation_id is None:
        return []

    with engine.connect() as connection:
        user_id = connection.execute(
            text("""
                SELECT id
                FROM app_users
                WHERE email = :email
            """),
            {"email": DEV_USER_EMAIL},
        ).scalar_one_or_none()

        if user_id is None:
            raise HTTPException(
                status_code=404,
                detail="Conversation not found.",
            )

        exists = connection.execute(
            text("""
                SELECT id
                FROM conversations
                WHERE id = :conversation_id
                  AND user_id = :user_id
            """),
            {
                "conversation_id": conversation_id,
                "user_id": user_id,
            },
        ).scalar_one_or_none()

        if exists is None:
            raise HTTPException(
                status_code=404,
                detail="Conversation not found.",
            )

        rows = connection.execute(
            text("""
                SELECT role, content
                FROM messages
                WHERE conversation_id = :conversation_id
                  AND role IN ('user', 'assistant')
                ORDER BY id DESC
                LIMIT 10
            """),
            {"conversation_id": conversation_id},
        ).mappings().all()

    # Restore chronological order.
    return list(reversed(rows))

def detect_period(question: str) -> str:
    q = question.lower()

    if re.search(r"\b(1y|1 year|one year|12 months)\b", q):
        return "1Y"

    if re.search(r"\b(6m|6 months|six months)\b", q):
        return "6M"

    if re.search(r"\b(1m|1 month|one month|30 days)\b", q):
        return "1M"

    return "3M"



def detect_intents(question: str) -> list[str]:
    q = question.lower()
    intents = []

    if any(word in q for word in [
        "undervalued", "overvalued", "valuation",
        "fundamental", "fundamentals", "balance sheet", "earnings",
    ]):
        intents.append("FUNDAMENTAL_RESEARCH")

    if re.search(r"\b(?:p/e|pe)\s+ratio\b", q):
        intents.append("FUNDAMENTAL_RESEARCH")

    if any(word in q for word in [
    ]):
        intents.append("FUNDAMENTAL_RESEARCH")

    if any(word in q for word in [
        "greedy algorithm", "greedy picks",
        "greedy selection", "algorithm experiment",
        "greedy portfolio",
    ]):
        intents.append("GREEDY_PICKS")

    if any(word in q for word in [
        "sector performance", "best performing sectors",
        "top performing sectors", "sector returns",
        "which sectors performed", "sectors",
    ]):
        intents.append("SECTOR_PERFORMANCE")

    if any(word in q for word in [
        "best performing", "top performing",
        "performed best", "top performers",
        "best performers", "top stocks",
        "best stocks", "highest returns",
        "highest return", "performance",
        "performers", "historical returns",
        "stock returns", "gainers", "most profitable",
    ]):
        intents.append("PERFORMANCE_STANDOUTS")

    if any(word in q for word in [
        "volatility", "drawdown", "sharpe",
        "risk", "stable", "stability",
        "least volatile", "lowest volatility",
    ]):
        intents.append("RISK_AND_STABILITY")

    education_concepts = [
        "volatility", "sharpe", "drawdown", "diversification",
        "correlation", "beta", "liquidity", "risk return",
        "risk-reward", "risk reward", "cagr",
    ]
    education_phrases = [
        "what is", "explain", "meaning of", "teach me",
        "beginner", "difference between", "how does",
        "why does", "help me understand",
    ]

    if (
        any(term in q for term in education_concepts)
        and any(phrase in q for phrase in education_phrases)
    ):
        intents.append("EDUCATION")

    if not intents and any(word in q for word in [
        "nifty 50", "what is nifty",
        "explain nifty", "stock market basics",
    ]):
        intents.append("EDUCATION")

    unique_intents = list(dict.fromkeys(intents))

    # Pure educational questions must not trigger market research.
    # Preserve mixed education + explicit stock research questions.
    if "EDUCATION" in unique_intents:
        research_actions = [
            "show", "list", "rank", "compare", "analyse",
            "analyze", "analysis of", "top performing",
            "best performing", "worst performing",
            "which stocks", "stocks with", "stock-wise",
            "performance of", "historical returns of",
            "least volatile stocks", "lowest volatility stocks",
            "portfolio", "sector performance",
        ]

        has_research_action = any(
            term in q for term in research_actions
        )

        if not has_research_action:
            return ["EDUCATION"]

    return unique_intents or ["GENERAL"]


def detect_intent(question: str) -> str:
    """Backward-compatible single-intent interface."""
    return detect_intents(question)[0]

def format_insights(ideas, metadata, category, comparison=None):
    categories = (
        [category] if isinstance(category, str)
        else list(category)
    )

    lines = []

    for current_category in categories:
        selected = [
            idea for idea in ideas
            if idea.get("category") == current_category
            and idea.get("status") == "AVAILABLE"
        ]

        if not selected:
            lines.append(
                f"{current_category}: No sufficient verified "
                "financial insight is available in this report."
            )
            lines.append("")
            continue

        lines.append(f"===== {current_category} =====")

        for index, idea in enumerate(selected, start=1):
            evidence = idea.get("evidence") or {}

            lines.append(
                f"{index}. {idea.get('title', 'Historical insight')}"
            )
            lines.append(
                f"What: {idea.get('what', 'Not available')}"
            )

            if idea.get("why"):
                lines.append(f"Why: {idea['why']}")

            fields = [
                ("period_return_pct", "Historical return", "%"),
                ("annualized_volatility_pct", "Annualised volatility", "%"),
                ("maximum_drawdown_pct", "Maximum drawdown", "%"),
                ("sharpe_ratio", "Sharpe ratio", ""),
                ("observations", "Return observations", ""),
            ]

            for key, label, suffix in fields:
                value = evidence.get(key)
                if value is not None:
                    lines.append(f"{label}: {value}{suffix}")

            if idea.get("risk"):
                lines.append(f"Risk: {idea['risk']}")

            lines.append("")

    required_categories = {
        "PERFORMANCE_STANDOUTS",
        "RISK_AND_STABILITY",
    }

    if comparison and required_categories.issubset(set(categories)):
        lines.extend([
            "DETERMINISTIC STOCK COMPARISON",
            "The following union and overlap were constructed by Python.",
            "Each stock appears once; category order is preserved.",
            "",
            "| Symbol | Return (%) | Volatility (%) | Drawdown (%) | Sharpe | Category |",
            "|---|---:|---:|---:|---:|---|",
        ])

        for stock in comparison.get("stocks", []):
            def cell(value, suffix=""):
                if value is None:
                    return "Not available"
                if isinstance(value, (int, float)):
                    return f"{value:.4f}{suffix}"
                return str(value)

            membership = " + ".join(stock.get("categories", []))

            lines.append(
                f"| {stock['symbol']} "
                f"| {cell(stock.get('period_return_pct'), '%')} "
                f"| {cell(stock.get('annualized_volatility_pct'), '%')} "
                f"| {cell(stock.get('maximum_drawdown_pct'), '%')} "
                f"| {cell(stock.get('sharpe_ratio'))} "
                f"| {membership} |"
            )

        overlap = comparison.get("overlap_symbols", [])
        overlap_text = ", ".join(overlap) if overlap else "None"

        lines.extend([
            "",
            f"Unique stocks in comparison: {comparison.get('stock_count', 0)}",
            f"Stocks appearing in both lists: {overlap_text}",
            "",
        ])

    lines.extend([
        "DATA METHODOLOGY",
        f"Index: {metadata.get('index')}",
        f"Period: {metadata.get('period')}",
        f"Historical start date: {metadata.get('requested_start_date')}",
        f"Historical end date: {metadata.get('common_end_date')}",
        f"Data as of: {metadata.get('data_as_of')}",
        f"Snapshot month: {metadata.get('snapshot_month')}",
        f"Price basis: {metadata.get('price_basis')}",
        (
            "Corporate-action coverage: "
            f"{metadata.get('corporate_action_coverage')}"
        ),
        f"Dividend adjusted: {metadata.get('dividend_adjusted')}",
        "",
        "IMPORTANT DISCLOSURES",
        "These are historical observations, not forecasts.",
        "The universe uses the latest monthly NIFTY 50 snapshot.",
        "Historical daily membership is not reconstructed.",
        "Corporate-action coverage is partial.",
    ])

    return "\\n".join(lines)


def build_financial_context(question):
    intents = detect_intents(question)

    if intents == ["FUNDAMENTAL_RESEARCH"]:
        return (
            "QUESTION CATEGORY: FUNDAMENTAL RESEARCH\\n\\n"
            "VERIFIED DATA STATUS:\\n"
            "Company fundamentals and valuation data are not yet "
            "integrated into GREEDY STOCKS.\\n"
            "Historical share-price performance cannot establish "
            "whether a company is undervalued or overvalued.\\n"
            "Do not invent financial statements, revenue, profit, "
            "debt, cash flow or valuation ratios.\\n"
            "You may explain what data would be needed for "
            "a proper fundamental analysis."
        )

    if intents == ["EDUCATION"]:
        return build_education_context(question)


    if intents == ["GENERAL"]:
        return (
            "QUESTION CATEGORY: GENERAL GREEDY STOCKS HELP\\n"
            "GREEDY STOCKS currently supports historical NIFTY 50 "
            "performance, volatility, drawdown, sector performance, "
            "historical Greedy Algorithm experiments and financial "
            "education. Fundamental valuation data is not yet integrated."
        )

    period = detect_period(question)

    try:
        report = generate_market_ideas(
            period=period,
            top_n=5,
        )

        verified_context = format_insights(
            ideas=report["ideas"],
            metadata=report["metadata"],
            category=intents,
            comparison=report.get("comparison"),
        )

        education_context = (
            build_education_context(
                question,
                ideas=report["ideas"],
                metadata=report["metadata"],
            )
            if "EDUCATION" in intents
            else ""
        )

        return (
            f"QUESTION CATEGORIES: {', '.join(intents)}\\n"
            f"REQUESTED PERIOD: {period}\\n\\n"
            "THE FOLLOWING IS VERIFIED APPLICATION DATA. "
            "Do not modify or invent its figures.\\n\\n"
            f"{verified_context}"
            + (
                "\\n\\n" + education_context
                if education_context else ""
            )
        )

    except Exception:
        logger.exception(
            "Financial intelligence generation failed."
        )
        return (
            "VERIFIED FINANCIAL DATA IS CURRENTLY UNAVAILABLE. "
            "Do not invent rankings, returns or risk metrics."
        )


def generate_gemini_answer(question, financial_context, history):
    api_key = os.getenv("GEMINI_API_KEY")

    if not api_key:
        logger.warning("GEMINI_API_KEY is not configured.")
        return None

    try:
        from google import genai

        client = genai.Client(api_key=api_key)

        # PHASE 2: Unified dependency-aware research planning.
        research_plan, planning_status = build_research_plan(
            client=client,
            model=GEMINI_MODEL,
            question=question,
            history=history,
        )

        handlers = {
            "market_research": lambda query: financial_context,
            "latest_price": lambda query: build_real_world_research_context(
                query, "latest_price"
            ),
            "affordability": lambda query: build_real_world_research_context(
                query, "affordability"
            ),
            "calculation": lambda query: build_calculation_context(
                client=client,
                model=GEMINI_MODEL,
                question=query,
            ),
            "historical_analysis": lambda query: (
                build_historical_intelligence_context(
                    client=client,
                    model=GEMINI_MODEL,
                    question=query,
                )
            ),
            "greedy_knowledge": lambda query: (
                build_greedy_knowledge_context(question=query)
            ),
            "greedy_v2_evidence": lambda query: (
                build_greedy_v2_evidence_context(query)
            ),
        }

        execution_log = execute_research_plan(
            plan=research_plan,
            handlers=handlers,
        )

        research_plan_context = format_execution_context(
            plan=research_plan,
            planning_status=planning_status,
            execution_log=execution_log,
        )

        def operation_result(operation_name, default):
            for entry in execution_log:
                if entry["operation"] == operation_name:
                    if entry["status"] == "COMPLETED":
                        return str(entry["result"])
                    return (
                        f"{default}\nOperation status: {entry['status']}\n"
                        f"Reason: {entry.get('reason', 'Unavailable')}"
                    )
            return default

        calculation_context = operation_result(
            "calculation",
            "No deterministic calculation was requested by the research plan.",
        )

        greedy_knowledge_context = operation_result(
            "greedy_knowledge",
            "No GREEDY source-code research was requested.",
        )

        historical_company_context = operation_result(
            "historical_analysis",
            "No company-specific historical analysis was requested.",
        )

        greedy_v2_evidence_context = operation_result(
            "greedy_v2_evidence",
            "No GREEDY V2 experiment research was requested.",
        )



        conversation_text = ""

        for item in history:
            role = item["role"].upper()
            content = item["content"]

            conversation_text += (
                f"\n{role}: {content}\n"
            )

        prompt = f"""
VERIFIED APPLICATION CONTEXT:

{financial_context}

UNIFIED PHASE 2 RESEARCH PLAN:

{research_plan_context}

COMPANY-SPECIFIC HISTORICAL INTELLIGENCE:
{historical_company_context}

GREEDY V2 EXPERIMENT AND WALK-FORWARD EVIDENCE:
{greedy_v2_evidence_context}

GREEDY SOURCE-CODE KNOWLEDGE:

{greedy_knowledge_context}

SOURCE KNOWLEDGE RULES:
- Use source excerpts to explain actual Greedy implementation.
- Distinguish Greedy V1, V2 and V3 only when the source supports it.
- Explain selection logic, objective functions, constraints and limitations
  only when visible in the supplied code.
- If a requested version or implementation is not found, say so.
- Never claim to have run an algorithm merely because its source was read.
- Never change algorithm behaviour or portfolio-generation decisions.
- Never invent source files, functions, metrics or algorithm features.


DETERMINISTIC CALCULATION ENGINE OUTPUT:

{calculation_context}

CALCULATION RULES:
- If status is CALCULATED, use the exact result and supplied inputs.
- Explain the formula, units and assumptions.
- Do not alter or invent calculated values.
- If no calculation was executed, do not claim one was verified.
- User-provided mathematical inputs do not require external market data.
- Distinguish mathematical examples from actual market observations.


RECENT CONVERSATION HISTORY:
{conversation_text if conversation_text else "No previous messages."}

CURRENT USER QUESTION:
{question}

INSTRUCTIONS:
Answer the current question directly.

Use the verified application context for all financial figures.
If the context says data is unavailable, do not invent it.
Use conversation history to understand follow-up questions.
Explain concepts in beginner-friendly language.
Keep the answer clear, useful and conversational.

CONTEXTUAL EDUCATION RULES:
- When contextual financial education is supplied, teach the concept first.
- Then connect it to the supplied historical stock metrics where available.
- Explain what the metric means in plain language before interpreting values.
- Never imply that low volatility means no risk or capital protection.
- Never imply that a high Sharpe ratio guarantees future performance.
- For diversification, correlation, beta, liquidity or CAGR, clearly state
  when the corresponding verified metric is unavailable.
- Do not invent benchmark comparisons, portfolio holdings or correlations.
- Use a small numerical example only if it is clearly labelled illustrative,
  not an actual GREEDY STOCKS observation.
- Do not convert an educational explanation into a buy/sell recommendation.

COMPARATIVE FINANCIAL RESEARCH RULES:
- When multiple research categories are requested, present each category separately.
- When two or more categories contain stock-level metrics, include a
  consolidated comparison table covering every available returned stock.
- The table should include stock symbol, historical return, annualised
  volatility, maximum drawdown, Sharpe ratio and category membership.
- Mark category membership as Performance, Low Volatility, Both, or the
  applicable verified category names.
- Identify overlap by matching exact stock symbols, not company-name guesses.
- Do not omit stocks merely to shorten the table.
- Use only figures present in VERIFIED APPLICATION CONTEXT.
- If a metric is unavailable, write "Not available"; never estimate it.
- Distinguish lowest volatility from lowest overall risk.
- Explain that Sharpe ratio measures excess return per unit of volatility.
  It does not prove investment skill or predict future performance.
- Do not describe historical rankings as investment recommendations.
- If only one category is requested, do not force an unnecessary comparison.
"""

        interaction = client.interactions.create(
            model=GEMINI_MODEL,
            input=prompt,
            system_instruction=SYSTEM_INSTRUCTION,
        )

        answer = interaction.output_text

        if not answer or not answer.strip():
            logger.warning("Gemini returned an empty response.")
            return None

        return answer.strip()

    except Exception:
        logger.exception("Gemini API request failed.")
        return None

# MAIN RESPONSE ENGINE
# --------------------------------------------------

def generate_greedy_answer(question, history):
    financial_context = build_financial_context(question)

    # Deterministic fallback is always available.
    fallback_answer = financial_context

    ai_answer = generate_gemini_answer(
        question=question,
        financial_context=financial_context,
        history=history,
    )

    if ai_answer:
        return ai_answer

    logger.warning(
        "Using deterministic fallback response."
    )

    return (
        "GREEDY AI is temporarily unavailable, so I'm "
        "showing the verified application information instead.\n\n"
        f"{fallback_answer}"
    )

@router.post("/chat", response_model=ChatResponse)
def send_message(request: ChatRequest):
    question = request.message.strip()

    if not question:
        raise HTTPException(
            status_code=400,
            detail="Message cannot be empty.",
        )

    try:
        # Validate existing conversation and load its history.
        history = get_recent_history(request.conversation_id)

        # Generate the answer before opening the write transaction.
        answer = generate_greedy_answer(
            question=question,
            history=history,
        )

        with engine.begin() as connection:
            user_id = get_or_create_dev_user(connection)

            if request.conversation_id is None:
                result = connection.execute(
                    text("""
                        INSERT INTO conversations (
                            user_id,
                            title
                        )
                        VALUES (
                            :user_id,
                            :title
                        )
                    """),
                    {
                        "user_id": user_id,
                        "title": question[:255],
                    },
                )

                conversation_id = result.lastrowid

            else:
                conversation_id = request.conversation_id

            connection.execute(
                text("""
                    INSERT INTO messages (
                        conversation_id,
                        role,
                        content
                    )
                    VALUES (
                        :conversation_id,
                        'user',
                        :content
                    )
                """),
                {
                    "conversation_id": conversation_id,
                    "content": question,
                },
            )

            connection.execute(
                text("""
                    INSERT INTO messages (
                        conversation_id,
                        role,
                        content
                    )
                    VALUES (
                        :conversation_id,
                        'assistant',
                        :content
                    )
                """),
                {
                    "conversation_id": conversation_id,
                    "content": answer,
                },
            )

            connection.execute(
                text("""
                    UPDATE conversations
                    SET updated_at = CURRENT_TIMESTAMP
                    WHERE id = :conversation_id
                """),
                {"conversation_id": conversation_id},
            )

        return ChatResponse(
            conversation_id=conversation_id,
            user_message=question,
            assistant_message=answer,
            created_at=datetime.now(timezone.utc),
        )

    except HTTPException:
        raise

    except Exception:
        logger.exception("Chat message processing failed.")

        raise HTTPException(
            status_code=500,
            detail="Chat service failed. Check backend logs.",
        )

@router.get("/chat/conversations")
def list_conversations(limit: int = 100):
    """
    Return the development user's recent conversations,
    newest activity first, with a short latest-message preview.
    """
    if limit < 1 or limit > 200:
        raise HTTPException(
            status_code=400,
            detail="Limit must be between 1 and 200.",
        )

    try:
        with engine.connect() as connection:
            user_id = connection.execute(
                text("""
                    SELECT id
                    FROM app_users
                    WHERE email = :email
                """),
                {"email": DEV_USER_EMAIL},
            ).scalar_one_or_none()

            if user_id is None:
                return {"conversations": []}

            rows = connection.execute(
                text("""
                    SELECT
                        c.id,
                        c.title,
                        c.created_at,
                        c.updated_at,
                        (
                            SELECT m.content
                            FROM messages AS m
                            WHERE m.conversation_id = c.id
                            ORDER BY m.id DESC
                            LIMIT 1
                        ) AS last_message,
                        (
                            SELECT COUNT(*)
                            FROM messages AS mc
                            WHERE mc.conversation_id = c.id
                        ) AS message_count
                    FROM conversations AS c
                    WHERE c.user_id = :user_id
                    ORDER BY c.updated_at DESC, c.id DESC
                    LIMIT :limit
                """),
                {
                    "user_id": user_id,
                    "limit": limit,
                },
            ).mappings().all()

            return {
                "conversations": [
                    dict(row) for row in rows
                ]
            }

    except Exception:
        logger.exception("Conversation list retrieval failed.")
        raise HTTPException(
            status_code=500,
            detail="Could not retrieve conversation history.",
        )

@router.get("/chat/conversations/{conversation_id}")
def get_conversation(conversation_id: int):
    try:
        with engine.connect() as connection:
            user_id = connection.execute(
                text("""
                    SELECT id
                    FROM app_users
                    WHERE email = :email
                """),
                {"email": DEV_USER_EMAIL},
            ).scalar_one_or_none()

            if user_id is None:
                raise HTTPException(
                    status_code=404,
                    detail="User not found.",
                )

            conversation = connection.execute(
                text("""
                    SELECT id, title, created_at, updated_at
                    FROM conversations
                    WHERE id = :conversation_id
                      AND user_id = :user_id
                """),
                {
                    "conversation_id": conversation_id,
                    "user_id": user_id,
                },
            ).mappings().first()

            if conversation is None:
                raise HTTPException(
                    status_code=404,
                    detail="Conversation not found.",
                )

            messages = connection.execute(
                text("""
                    SELECT id, role, content, created_at
                    FROM messages
                    WHERE conversation_id = :conversation_id
                    ORDER BY id ASC
                """),
                {"conversation_id": conversation_id},
            ).mappings().all()

            return {
                "conversation": dict(conversation),
                "messages": [dict(message) for message in messages],
            }

    except HTTPException:
        raise

    except Exception:
        logger.exception("Conversation retrieval failed.")

        raise HTTPException(
            status_code=500,
            detail="Could not retrieve conversation.",
        )

@router.delete("/chat/conversations/{conversation_id}")
def delete_conversation(conversation_id: int):
    """
    Delete one conversation belonging to the development user
    and all of its messages.
    """
    try:
        with engine.begin() as connection:
            user_id = connection.execute(
                text("""
                    SELECT id
                    FROM app_users
                    WHERE email = :email
                """),
                {"email": DEV_USER_EMAIL},
            ).scalar_one_or_none()

            if user_id is None:
                raise HTTPException(
                    status_code=404,
                    detail="Conversation not found.",
                )

            exists = connection.execute(
                text("""
                    SELECT id
                    FROM conversations
                    WHERE id = :conversation_id
                      AND user_id = :user_id
                """),
                {
                    "conversation_id": conversation_id,
                    "user_id": user_id,
                },
            ).scalar_one_or_none()

            if exists is None:
                raise HTTPException(
                    status_code=404,

                """),

                },
            )

            "conversation_id": conversation_id,
            "message": "Conversation deleted successfully.",
